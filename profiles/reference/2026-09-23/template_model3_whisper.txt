# ============================================================
#
# ЯЗЫК: Арамейский/Древнееврейский (Вавилонская традиция чтения)
# ПЕРИОД: до VI-VIII в. н.э.
# ТИП ТЕКСТА: Религиозный/молитвенный гимн
#
# ============================================================

# КЛЮЧЕВЫЕ ФОНЕТИЧЕСКИЕ ОТЛИЧИЯ от современного иврита/арамейского:
#
# 1. ЗВУК [w] ВМЕСТО [v]:
#    - вав (ו) = [w] (как английское w), НЕ [v]
#    - бет (ב) после гласного = [w] или [β] (губы внутрь), НЕ [v]
#    - "ав" (Отец) = "оув" [ʔow], не "ав"
#    - "ве" (и) = "ууве" [uːwe], не "ве"
#    - "бар" (сын) = "вер" [war], не "бар"
#
# 2. КАМАЦ = [o], НЕ [a]:
#    - Вавилонская традиция: камац = [o] или [ɔ] (округлённый)
#    - דָּבָר = "довор", не "давар"
#    - Элоhим: гласные ближе к [o]/[e], не [a]
#
# 3. ПАЛАТАЛИЗАЦИЯ:
#    - ד перед [i] → "дь" [dʲ]: "дьин" (суд), не "дин"
#    - ר перед [u]/[o] → "рю" [rʲ]: "рюх" (дух), не "рух"
#    - כ в неоарамейских → "ч" [tʃ]
#
# 4. ПРОТЕТИЧЕСКИЙ ГЛАСНЫЙ:
#    - Перед алефом и вавом вставляется гласный [o] или [uː]
#    - "ав" → "оув" (добавочный о + w)
#    - "ве" → "ууве" (добавочный уу + w)
#
# 5. יָהּ (Ях) = "Ю" [ju]:
#    - Лабиализация: [ja] → [ju]
#    - "Элоhим Ю!" = "Бог — Я!", но звучит как "Ю"
#    - НЕ "Я", а "Ю"
#
# 6. ШИН = [ɕ] (русское "щ"):
#    - שׁ = [ɕ] в быстрой речи (с примесью [s])
#    - "шалом" может звучать как "щалом"
#    - "шва" может звучать как "щва" (с переходом с→ш→щ)
#
# 7. НЕОПРЕДЕЛЁННЫЕ ГЛАСНЫЕ:
#    - Гласные часто ПРОМЕЖУТОЧНЫЕ между стандартными:
#      * между "а" и "я" → один слитный звук
#      * между "и" и "ы"
#      * между "у" и "о"
#      * между "е" и "э"
#    - Это НЕ редукция и НЕ ошибка — это система
#    - Модель может слышать "и" или "ы" — это ОДИН звук между ними
#
# 8. ДОЛГИЕ ГЛАСНЫЕ:
#    - [uː] → "уу" (двойное у, тянется)
#    - [aː] → "аа"
#    - [oː] → "оо"
#
# 9. ПРЕДЫХАНИЕ:
#    - Перед словом "Элоhим" возможен придыхательный [h]
#    - Обозначается как [предых] перед словом
#
# 10. МАЛЫЕ ПАУЗЫ:
#    - Короткие паузы внутри фраз (не между словами, а внутри)
#    - Обозначаются как ()
#    - Модель не должна воспринимать как конец слова

# СЛОВАРЬ ОЖИДАЕМЫХ СЛОВ:
# Элоhим — Бог (с предыханием, h придыхательный)
# шалом — мир (шин может быть [ɕ])
# ав — Отец (произносится "оув", вав=[w])
# рух — дух (произносится "рюх", р йотированное)
# Ешуа — Иисус (полная арамейская форма)
# Иешо — Иисус (краткая форма)
# хос/Ихос — Иисус (возможно греческая форма)
# дин — суд (произносится "дьин", д палатализованное)
# бар — сын (произносится "вер/уувер", вав=[w])
# дэ — который (арамейский относительный союз)
# са/ша — который (относительный союз)
# ве — и (произносится "ууве", вав=[w])
# дор — поколение
# дэйа — знание
# шва — пустота/напрасно
# йеш — существует
# Ю — Я (краткое имя Бога, лабиализованное)
# ни — мой (притяжательное 1 л.)
# лахем — хлеб
# фи/пэ — уста
# ла/ло — не
# равви — учитель мой

# СТРУКТУРНЫЕ ПАТТЕРНЫ ТЕКСТА:
# - Элоhим встречается в каждой строке (рефрен)
# - шалом в большинстве строк
# - Ю (имя Бога) как восклицание
# - рух (дух) + дэ-ни (мой)
# - са/ша (который) перед существительными
# - ве (и) как соединительный союз
# - Темы: Отец → хлеб → Бог → мир → Сущий → знание → дух → Иисус → суд → поколение

# ПРЕДУПРЕЖДЕНИЯ ДЛЯ МОДЕЛИ:
# - НЕ заменять [w] на [v]
# - НЕ заменять [o] (камац) на [a]
# - НЕ игнорировать палатализацию (дь, рю)
# - НЕ игнорировать протетические гласные (оу, уу)
# - НЕ разделять неопределённые гласные на два звука
# - НЕ заменять "Ю" на "Я"
# - Учитывать малые паузы внутри слов
# - Учитывать долгие гласные (уу, аа, оо)
