# IPA-ТАБЛИЦА СОГЛАСНЫХ:
# [ʔ]  алеф    — глоттальная остановка (часто нулевая, с протетическим гласным)
# [b]  бет-1   — взрывной (начало слова, до спирантизации)
# [w]  бет-2   — лабиовелярный (после гласного, ДРЕВНЕЕ произношение)
# [β]  бет-3   — билабиальный фрикатив (альтернатива, нижняя губа внутрь)
# [ɡ]  гимель  — взрывной
# [d]  далет   — взрывной
# [dʲ] далет-п — палатализованный ПЕРЕД [i] /iː/
# [ð]  далет-с — спирантизованный (позже, не наша традиция)
# [h]  hей     — глоттальный
# [w]  вав     — лабиовелярный (ДРЕВНЕЕ, не [v]!)
# [z]  заин    — свистящий
# [ħ]  хет     — фарингальный глухой
# [tˤ] тет     — эмфатический
# [j]  йод     — палатальный
# [k]  каф     — взрывной
# [tʃ] каф-п   — палатализованный (неоарамейские диалекты)
# [l]  ламед   — латеральный
# [m]  мем     — носовой
# [n]  нун     — носовой
# [s]  самех   — свистящий
# [ʕ]  аин     — фарингальный звонкий
# [p]  пе      — взрывной
# [f]  пе-с    — спирантизованный
# [sˤ] цаде    — эмфатический свистящий
# [q]  коф     — увулярный
# [r]  реш     — альвеолярный
# [rʲ] реш-й   — йотированный (ПЕРЕД [u]/[o]/[oː])
# [ʃ]  шин     — постальвеолярный
# [ɕ]  шин-щ   — палатоальвеолярный (русское "щ", в быстрой речи)
# [s]  син     — свистящий (иногда = самех)

# IPA-ТАБЛИЦА ГЛАСНЫХ (Вавилонская традиция):
# [o]    камац    — задний округлённый (Вавилонская, НЕ [a]!)
# [ɔ]    камац-т  — полуоткрытый задний (Тиверийский = [ɔ], Вавилонский = [o])
# [a]    патах    — открытый передний
# [eː]   цере     — долгий полуоткрытый
# [ɛ]    сегол    — краткий (Вавилонский = [a], слит с патах)
# [i]    хирек    — закрытый передний
# [oː]   холам    — долгий задний (Вавилонский иногда = [eː]!)
# [uː]   шурек   — долгий задний (Вавилонский иногда = [y]!)
# [u]    кубуц    — краткий задний (Вавилонский = [y]!)
# [ə]    шва      — редуцированный (Вавилонский = [a] или ноль)

# ПРОМЕЖУТОЧНЫЕ (НЕОПРЕДЛЁННЫЕ) ГЛАСНЫЕ:
# [a]~[ja]  — между "а" и "я" (алофон, не пауза!)
# [i]~[ɨ]   — между "и" и "ы"
# [u]~[o]   — между "у" и "о"
# [o]~[u]~[jo] — тройной с йотацией
# [e]~[ɛ]   — между "е" и "э"

# ПРОТЕТИЧЕСКИЕ ГЛАСНЫЕ:
# Перед алефом: [ʔo] или [ʔu] — вставной гласный
# Перед вавом: [uː] — долгий протетический, плавно в [w]
# Примеры: אָב [ʔow] не [ʔav]; וְ [uːwe] не [ve]

# ЙОТАЦИЯ И ПАЛАТАЛИЗАЦИЯ:
# ד + [i] → [dʲi] — "дьин" для דִּין
# ר + [u]/[o] → [rʲu]/[rʲo] — "рюх" для רוּחַ
# כ + [e]/[i] → [tʃe]/[tʃi] — палатализация каф (в неоарамейских)

# ЛАБИАЛИЗАЦИЯ יָהּ:
# [ja] → [ju] — "Ю" вместо "Я" (возможное влияние вав=[w] как огубление)

# ДЛИНА:
# Долгий гласный: [uː], [aː], [oː] — удвоение в записи
# Долгий согласный: [mː], [lː] — удвоение в записи

# СИСТЕМА ОБОЗНАЧЕНИЙ ДЛЯ ЗАПИСИ:
# ( )      — малая пауза
# (а~я)    — неопределённый гласный между а и я (ОДИН звук)
# (и~ы)    — между и и ы
# (у~о)    — между у и о
# (о~у~ё)  — тройной, с йотацией
# (е~э)    — между е и э
# уу, аа   — долгий гласный
# мм, лл   — долгий согласный
# [предых] — придыхание перед словом
# дь       — палатализованный д (перед и)
# рю       — йотированный р (перед у/о)

# КЛЮЧЕВЫЕ СЛОВА (IPA):
# Элоhим:  [ʔeloːhim] или [ʔɛloːħim] (с придыханием)
# шалом:   [ɕalom] или [ʃalom]
# ав:      [ʔow] → "оув"
# рух:     [rʲux] → "рюх"
# Ешуа:    [jeʃuːa] → "ешуа"
# дин:     [dʲin] → "дьин"
# бар:     [war] → "вер/уувер"
# дэ:      [de] → "дэ"
# ве:      [uːwe] → "ууве"
# дор:     [dor] → "дор"
# дэйа:    [deːja] → "дэйа"
# шва:     [ɕwa] → "шва"
# йеш:     [jeʃ] → "йеш"
# Ю:       [ju] → "ю" (лабиализованное יָהּ)
# са:      [ʃa] → "са" (относительный союз)
# ни:      [ni] → "ни" (притяжательное 1 л.)
# лахем:   [lahem] → "лахем" (хлеб)
# фи/пэ:   [pi]/[pe] → "фи/пэ" (уста)
