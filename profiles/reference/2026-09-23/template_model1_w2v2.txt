# ОЖИДАЕМЫЕ ЗВУКИ И ИХ IPA:

# СОГЛАСНЫЕ:
# א (алеф)   → [ʔ] — глоттальная остановка, часто протетический гласный перед ним
# ב (бет)   → [b] в начале слова, [w]/[β] после гласного (ДО спирантизации — всегда [b] или [w])
# ג (гимель) → [ɡ], в некоторых диалектах палатализуется → [dʒ]
# ד (далет)  → [d], после гласного [ð] (позже), ПЕРЕД [i] → [dʲ] (палатализация!)
# ה (hей)   → [h]
# ו (вав)   → [w] (ДРЕВНЕЕ), НЕ [v]! Современное [v] — позднее
# ז (заин)  → [z]
# ח (хет)   → [ħ] — глоттальный фрикатив
# ט (тет)   → [tˤ] — эмфатический
# י (йод)   → [j]
# כ (каф)   → [k], в некоторых диалектах [tʃ] (палатализация в неоарамейских)
# ל (ламед) → [l]
# מ (мем)   → [m]
# נ (нун)   → [n]
# ס (самех) → [s]
# ע (аин)   → [ʕ] — фарингальный
# פ (пе)    → [p], после гласного [f]
# צ (цаде)  → [sˤ] — эмфатический
# ק (коф)   → [q] — увулярный
# ר (реш)   → [r], ПЕРЕД округлённым гласным → [rʲ] (йотированный!)
# שׁ (шин)  → [ʃ], в быстрой речи → [ɕ] (русское щ) с примесью [s]
# שׂ (син)  → [s], иногда сливается с самех
# ת (тав)   → [t], после гласного [θ] (позже)

# ГЛАСНЫЕ:
# קָמַץ (камац)  → [o] или [ɔ] (В ВАВИЛОНСКОЙ ТРАДИЦИИ!), НЕ [a]!
# פַּתַח (патах) → [a]
# צֵרֵ (цере)    → [eː]
# סֶגוֹל (сегол) → [ɛ] (в Вавилонской = [a], слит с патах)
# חִירִק (хирек) → [i]
# חוֹלַם (холам) → [oː] (в некоторых Вавилонских = [eː]!)
# שׁוּרֶק (шурек)→ [uː] (в Вавилонском иногда = [y]!)
# קֻבּוּץ (кубуц)→ [u] (в Вавилонском = [y]!)
# шва (шва)      → [ə] (в Вавилонской = [a] или ноль)

# ОСОБЫЕ ЯВЛЕНИЯ:
# 1. ПРОТЕТИЧЕСКИЙ ГЛАСНЫЙ перед алефом и вавом: [ʔo], [uː] перед [w]
#    Пример: אָב (ав) → [ʔow] → "оув"; וְ (ве) → [uːwe] → "ууве"
# 2. ПАЛАТАЛИЗАЦИЯ: ד перед [i] → [dʲ] ("дь"); ר перед [u]/[o] → [rʲ] ("рю")
# 3. НЕОПРЕДЕЛЁННЫЕ ГЛАСНЫЕ: промежуточные звуки между стандартными
#    [a]~[ja] — между "а" и "я"
#    [i]~[ɨ] — между "и" и "ы"
#    [u]~[o] — между "у" и "о"
#    [o]~[u]~[jo] — тройной промежуточный с йотацией
#    [e]~[ɛ] — между "е" и "э"
# 4. יָהּ (Ях) → [ju] ("Ю") — лабиализация, не [ja]!
# 5. ДОЛГИЕ ГЛАСНЫЕ: удвоение: [uː] → "уу", [aː] → "аа"
# 6. ПРЕДЫХАНИЕ перед Элоhим: [h] придыхание

# КЛЮЧЕВЫЕ СЛОВА ДЛЯ РАСПОЗНАВАНИЯ:
# Элоhим /ʔeloːhim/ → "эло-h-им" (с предыханием)
# шалом /ʃalom/ → "шалом" (шин может быть [ɕ])
# ав /ʔow/ → "оув" (вав=[w], протетический гласный)
# рух /rʲux/ → "рюх" (йотированное р, призвук о)
# Ешуа /jeʃuːa/ → "ешуа" (полная форма)
# Иешо /jeʃo/ → "иешо" (краткая форма)
# дин /dʲin/ → "дьин" (палатализация д перед и)
# бар /war/ → "вер/вар" (вав=[w], протетический гласный)
# дэ /de/ → "дэ" (арамейский относительный союз)
# са /ʃa/ → "са/ша" (относительный союз)
# ве /we/ → "ве" (вав=[w], возможен протетический гласный)
# дор /dor/ → "дор" (поколение)
# дэйа /deːja/ → "дэйа" (знание)
# шва /ʃwa/ → "шва" (пустота, шин=[ɕ] с примесью [s])
# йеш /jeʃ/ → "йеш" (существует)
# Ю /ju/ → "ю" (краткое имя Бога, лабиализованное)

# ФОНЕТИЧЕСКИЕ ПРАВИЛА ДЛЯ РАСПОЗНАВАНИЯ:
# 1. [w] вместо [v] везде (вав, бет после гласного)
# 2. [o]/[ɔ] вместо [a] для камац (Вавилонская традиция)
# 3. [dʲ] перед [i] (палатализация)
# 4. [rʲ] перед [u]/[o] (йотация)
# 5. Протетический гласный перед алефом и вавом
# 6. [ɕ] (русское щ) вместо [ʃ] для шин в быстрой речи
# 7. [ju] вместо [ja] для יָהּ
# 8. Долгие гласные удваиваются
# 9. Неопределённые/промежуточные гласные — НОРМА, не ошибка
